    ```python
    """Compare all Phase 4 experimental stages and produce wave_comparison outputs."""
    import json, os

    CLASS_NAMES = [
        "tench", "english_springer", "cassette_player", "chain_saw", "church",
        "french_horn", "garbage_truck", "gas_pump", "golf_ball", "parachute",
    ]

    # Load all results
    baseline = json.load(open("results/baseline_vgg16.json"))
    stage_a = json.load(open("results/stageA_full.json"))
    exp1 = json.load(open("results/exp1_full.json"))
    exp2 = json.load(open("results/exp2_full.json"))

    # Optional: Phase 3 config (no training results, just param count)
    phase3_config = None
    if os.path.exists("results/sgnnet_config.json"):
        phase3_config = json.load(open("results/sgnnet_config.json"))
    ```

    **Build wave_comparison.json:**
    ```python
    comparison = {
        "vgg16_baseline": {
            "model": "VGG16 FC (frozen)",
            "params": baseline["fc_params"],
            "top1_accuracy": baseline["top1_accuracy"],
            "mAP": baseline["mAP"],
        },
        "stageA": {
            "model": "Stage A (static binary C, no dynamic)",
            "params": stage_a["params"],
            "percent_of_vgg16_fc": stage_a["percent_of_vgg16_fc"],
            "top1_accuracy": stage_a["top1_accuracy"],
            "mAP": stage_a["mAP"],
            "per_class": stage_a["per_class"],
            "hyperparams": stage_a["hyperparams"],
        },
        "exp1": {
            "model": "Exp 1 (spatial path-length phase)",
            "params": exp1["params"],
            "percent_of_vgg16_fc": exp1["percent_of_vgg16_fc"],
            "top1_accuracy": exp1["top1_accuracy"],
            "mAP": exp1["mAP"],
            "per_class": exp1["per_class"],
            "hyperparams": exp1["hyperparams"],
        },
        "exp2": {
            "model": "Exp 2 (spatial phase + W_phase)",
            "params": exp2["params"],
            "percent_of_vgg16_fc": exp2["percent_of_vgg16_fc"],
            "top1_accuracy": exp2["top1_accuracy"],
            "mAP": exp2["mAP"],
            "per_class": exp2["per_class"],
            "hyperparams": exp2["hyperparams"],
        },
        "deltas": {
            "exp1_vs_stageA": {
                "top1_delta": exp1["top1_accuracy"] - stage_a["top1_accuracy"],
                "mAP_delta": exp1["mAP"] - stage_a["mAP"],
                "param_delta": exp1["params"] - stage_a["params"],
            },
            "exp2_vs_exp1": {
                "top1_delta": exp2["top1_accuracy"] - exp1["top1_accuracy"],
                "mAP_delta": exp2["mAP"] - exp1["mAP"],
                "param_delta": exp2["params"] - exp1["params"],
            },
        },
    }
    with open("results/wave_comparison.json", "w") as f:
        json.dump(comparison, f, indent=2)
    ```

    **Build wave_comparison.md:**
    ```python
    md_lines = [
        "# Wave Architecture Comparison",
        "",
        "## Aggregate Metrics",
        "",
        "| Model | Params | % VGG FC | Top-1 | mAP | Key addition |",
        "|-------|--------|----------|-------|-----|--------------|",
        f"| VGG16 FC (frozen) | {baseline['fc_params']:,} | 100% | {baseline['top1_accuracy']:.4f} | {baseline['mAP']:.4f} | Reference |",
    ]

    if phase3_config:
        md_lines.append(f"| SGNNET v1 (Phase 3) | {phase3_config['total_params']:,} | {phase3_config['percent_of_vgg16_fc']:.2f}% | -- | -- | Learned C, amplitude routing |")

    md_lines.extend([
        f"| Stage A (static) | {stage_a['params']:,} | {stage_a['percent_of_vgg16_fc']}% | {stage_a['top1_accuracy']:.4f} | {stage_a['mAP']:.4f} | Binary C, no dynamic |",
        f"| Exp 1 (spatial phase) | {exp1['params']:,} | {exp1['percent_of_vgg16_fc']}% | {exp1['top1_accuracy']:.4f} | {exp1['mAP']:.4f} | + proximity w/ path-length phase |",
        f"| Exp 2 (spatial + W_phase) | {exp2['params']:,} | {exp2['percent_of_vgg16_fc']}% | {exp2['top1_accuracy']:.4f} | {exp2['mAP']:.4f} | + learned phase operator |",
        "",
        "## Contribution Analysis",
        "",
        f"- **Exp 1 vs Stage A:** top1 delta = {exp1['top1_accuracy'] - stage_a['top1_accuracy']:+.4f}, mAP delta = {exp1['mAP'] - stage_a['mAP']:+.4f}",
        f"  - Proximity routing with spatial phase {'improves' if exp1['mAP'] > stage_a['mAP'] else 'does not improve'} over static wiring",
        f"- **Exp 2 vs Exp 1:** top1 delta = {exp2['top1_accuracy'] - exp1['top1_accuracy']:+.4f}, mAP delta = {exp2['mAP'] - exp1['mAP']:+.4f}",
        f"  - W_phase adds {exp2['params'] - exp1['params']} parameters",
        f"  - Learned phase operator {'improves' if exp2['mAP'] > exp1['mAP'] else 'does not improve'} over geometry-only phase",
        "",
        "## Per-Class Comparison",
        "",
        "| Class | VGG16 Acc | Stage A Acc | Exp1 Acc | Exp2 Acc | Exp1 AP | Exp2 AP |",
        "|-------|-----------|-------------|----------|----------|---------|---------|",
    ])

    for cls in CLASS_NAMES:
        vgg_acc = baseline.get("per_class", {}).get(cls, {}).get("accuracy", "--")
        sa_acc = stage_a["per_class"].get(cls, {}).get("accuracy", "--")
        e1_acc = exp1["per_class"].get(cls, {}).get("accuracy", "--")
        e2_acc = exp2["per_class"].get(cls, {}).get("accuracy", "--")
        e1_ap = exp1["per_class"].get(cls, {}).get("AP", "--")
        e2_ap = exp2["per_class"].get(cls, {}).get("AP", "--")
        # Format floats
        fmt = lambda v: f"{v:.4f}" if isinstance(v, float) else str(v)
        md_lines.append(f"| {cls} | {fmt(vgg_acc)} | {fmt(sa_acc)} | {fmt(e1_acc)} | {fmt(e2_acc)} | {fmt(e1_ap)} | {fmt(e2_ap)} |")

    md_lines.extend(["", "## Hyperparameters Used", ""])
    for name, data in [("Stage A", stage_a), ("Exp 1", exp1), ("Exp 2", exp2)]:
        hp = data["hyperparams"]
        md_lines.append(f"**{name}:** K={hp['K']}, N_hidden={hp['N_hidden']}, lr_Wpos={hp['lr_Wpos']:.2e}, lambda_safety={hp['lambda_safety']:.3f}, batch_size={hp['batch_size']}")

    with open("results/wave_comparison.md", "w") as f:
        f.write("\n".join(md_lines) + "\n")

    print("Saved results/wave_comparison.json and results/wave_comparison.md")
    ```
