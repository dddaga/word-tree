    **Imports:**
    ```python
    """Verify stored soft labels match VGG16 direct evaluation."""
    from __future__ import annotations
    import json
    from pathlib import Path
    import numpy as np
    from src.data.store import TensorStore
    from src.data.dataset import IMAGENETTE_CLASSES
    ```

    **Main flow (inside `if __name__ == "__main__":`):**

    1. **Load baseline results:**
       ```python
       baseline_path = Path("results/baseline_vgg16.json")
       with open(baseline_path) as f:
           baseline = json.load(f)
       direct_acc = baseline["top1_accuracy"]
       ```

    2. **Load stored soft labels from HDF5:**
       ```python
       store = TensorStore("data/store.h5")
       _, soft_labels, labels = store.get_split("val")
       # soft_labels: np.ndarray [3925, 10], labels: np.ndarray [3925]
       ```

    3. **Compute argmax accuracy from stored soft labels:**
       ```python
       preds = np.argmax(soft_labels, axis=1)
       store_acc = float(np.mean(preds == labels))
       ```

    4. **Check accuracy match (per D-03: within 0.1% = 0.001):**
       ```python
       acc_diff = abs(store_acc - direct_acc)
       accuracy_match = acc_diff <= 0.001
       print(f"Direct eval accuracy: {direct_acc:.6f}")
       print(f"Stored soft-label argmax accuracy: {store_acc:.6f}")
       print(f"Difference: {acc_diff:.6f} ({'PASS' if accuracy_match else 'FAIL'} <= 0.001)")
       ```

    5. **Check entropy (per D-03: warn if mean entropy < 0.01 nats):**
       ```python
       eps = 1e-10
       entropy = -np.sum(soft_labels * np.log(soft_labels + eps), axis=1)
       mean_entropy = float(np.mean(entropy))
       print(f"Mean soft-label entropy: {mean_entropy:.4f} nats")
       if mean_entropy < 0.01:
           print("WARNING: Mean entropy < 0.01 nats — distribution may be collapsed to near one-hot")
       ```

    6. **Check class balance (per D-03: warn if any class < 5% of val set):**
       ```python
       n_val = len(labels)
       for i, name in enumerate(IMAGENETTE_CLASSES):
           count = int(np.sum(labels == i))
           pct = count / n_val * 100
           if pct < 5.0:
               print(f"WARNING: Class '{name}' has only {pct:.1f}% of val set ({count}/{n_val})")
       ```

    7. **Update baseline JSON with check result (per D-03: true if accuracy_match):**
       ```python
       baseline["soft_label_accuracy_check"] = accuracy_match
       baseline["soft_label_accuracy_diff"] = acc_diff
       baseline["soft_label_mean_entropy"] = mean_entropy
       with open(baseline_path, "w") as f:
           json.dump(baseline, f, indent=2)
       print(f"\nUpdated {baseline_path} with soft_label_accuracy_check={accuracy_match}")
       ```
